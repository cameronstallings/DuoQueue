/**
 * The pure half of the swipe pager the paywall and the tour share (usePager.ts, Pager.tsx,
 * PagerDots.tsx). Relative imports only, so vitest runs it in node. The "worklet" directives let
 * the dots and the tour's foot call dotLight and footFade on the UI thread; in node they are
 * plain strings.
 */

/** The one-time peek: how far the pager slides to show the next page's edge, when, and for how long. */
export const PEEK_OFFSET = 32;
export const PEEK_DELAY_MS = 1400;
export const PEEK_HOLD_MS = 450;

/** One dot spec for both pagers (check.md #9): 7pt dots, 8pt apart, whose fill follows the finger. */
export const DOT_SIZE = 7;
export const DOT_GAP = 8;

export function clampPage(index: number, count: number): number {
  if (count <= 0) return 0;
  return Math.max(0, Math.min(count - 1, Math.round(index)));
}

/** The page nearest a horizontal offset, clamped past the ends. */
export function pageAtOffset(x: number, width: number, count: number): number {
  if (width <= 0) return 0;
  return clampPage(x / width, count);
}

/** How lit dot `index` is at offset `x`: 1 on its own page, 0 a page away, linear between. */
export function dotLight(x: number, index: number, width: number): number {
  "worklet";
  if (width <= 0) return index === 0 ? 1 : 0;
  return Math.max(0, 1 - Math.abs(x / width - index));
}

/**
 * The tour's foot. Skip fades out over the first half of the last turn and Start swiping fades
 * in over the second half, so the two are never both half visible (found by rendering the
 * mid-swipe frame, design-tour.md).
 */
export function footFade(x: number, lastIndex: number, width: number): { skip: number; start: number } {
  "worklet";
  if (lastIndex <= 0) return { skip: 0, start: 1 };
  if (width <= 0) return { skip: 1, start: 0 };
  const lastX = lastIndex * width;
  const half = width / 2;
  const skip = Math.max(0, Math.min(1, (lastX - half - x) / half));
  const start = Math.max(0, Math.min(1, (x - (lastX - half)) / half));
  return { skip, start };
}

export interface PeekInputs {
  count: number;
  /** Page one has its content (a peek at an empty page teaches nothing). */
  ready: boolean;
  peeked: boolean;
  touched: boolean;
  reduceMotion: boolean;
  screenReader: boolean;
}

/**
 * Once, only with a next page to show, only once page one has content, and never under Reduce
 * Motion, under a screen reader (the dots say "Page 1 of 3" instead, and a pager that moves by
 * itself mid-sentence is an interruption), or once the user has moved the pager themselves.
 */
export function shouldPeek(i: PeekInputs): boolean {
  return i.count > 1 && i.ready && !i.peeked && !i.touched && !i.reduceMotion && !i.screenReader;
}

/** The dot row's accessibilityValue: VoiceOver reads "Page, 1 of 3". */
export function pagerA11yValue(index: number, count: number): { min: number; max: number; now: number; text: string } {
  const max = Math.max(count, 1);
  const now = clampPage(index, count) + 1;
  return { min: 1, max, now, text: `${now} of ${max}` };
}

/** VoiceOver's swipe up or down on the dots: one page per action, clamped at the ends. */
export function adjustedPage(action: string, index: number, count: number): number {
  if (action === "increment") return clampPage(index + 1, count);
  if (action === "decrement") return clampPage(index - 1, count);
  return clampPage(index, count);
}

import { useEffect, useState } from "react";
import { AccessibilityInfo } from "react-native";
import { useReducedMotion } from "react-native-reanimated";

/**
 * Reduce Motion, live. Reanimated's useReducedMotion() is read once at launch; this starts from
 * it (so the first frame is right) and follows the switch while the screen is open, as
 * Previously's pagers do.
 */
export function useReduceMotionLive(): boolean {
  const atLaunch = useReducedMotion();
  const [reduce, setReduce] = useState(atLaunch);

  useEffect(() => {
    let alive = true;
    void AccessibilityInfo.isReduceMotionEnabled().then((value) => {
      if (alive) setReduce(value);
    });
    const subscription = AccessibilityInfo.addEventListener("reduceMotionChanged", (value) => setReduce(value));
    return () => {
      alive = false;
      subscription.remove();
    };
  }, []);

  return reduce;
}

/** Whether VoiceOver (or TalkBack) is running, so the peek can stand down for it. */
export function useScreenReader(): boolean {
  const [on, setOn] = useState(false);

  useEffect(() => {
    let alive = true;
    void AccessibilityInfo.isScreenReaderEnabled().then((value) => {
      if (alive) setOn(value);
    });
    const subscription = AccessibilityInfo.addEventListener("screenReaderChanged", (value) => setOn(value));
    return () => {
      alive = false;
      subscription.remove();
    };
  }, []);

  return on;
}

import { useCallback, useEffect, useRef, useState } from "react";
import { type NativeScrollEvent, type NativeSyntheticEvent, useWindowDimensions } from "react-native";
import Animated, {
  type AnimatedRef,
  type SharedValue,
  useAnimatedRef,
  useAnimatedScrollHandler,
  useSharedValue,
} from "react-native-reanimated";

import { PEEK_DELAY_MS, PEEK_HOLD_MS, PEEK_OFFSET, adjustedPage, clampPage, pageAtOffset, shouldPeek } from "./pagerMath";
import { useReduceMotionLive, useScreenReader } from "./useAccessibilityFlags";

export interface PagerController {
  count: number;
  width: number;
  /** The page the pager has settled on: a swipe's end, the dots, the chevron. */
  page: number;
  /** The live horizontal offset on the UI thread. The dots and the tour's foot follow it. */
  scrollX: SharedValue<number>;
  ref: AnimatedRef<Animated.ScrollView>;
  goTo: (index: number) => void;
  /** VoiceOver's increment/decrement on the dots. */
  adjust: (action: string) => void;
  scrollHandler: ReturnType<typeof useAnimatedScrollHandler>;
  onScrollBeginDrag: () => void;
  onMomentumScrollEnd: (event: NativeSyntheticEvent<NativeScrollEvent>) => void;
}

/**
 * One swipe pager's state, shared by the paywall and the tour. Previously's pattern
 * (previously/app/paywall.tsx, tour.tsx) on a Reanimated scroll offset instead of RN Animated.
 */
export function usePager(count: number, options: { peekReady: boolean }): PagerController {
  const { width } = useWindowDimensions();
  const ref = useAnimatedRef<Animated.ScrollView>();
  const scrollX = useSharedValue(0);
  const [page, setPage] = useState(0);
  const reduceMotion = useReduceMotionLive();
  const screenReader = useScreenReader();
  // Refs, not state: the peek's timers read them, and neither should re-render the screen.
  const touched = useRef(false);
  const peeked = useRef(false);

  const goTo = useCallback(
    (index: number) => {
      // Any turn the user makes (chevron, dots) counts as touching the pager, so the peek can
      // never slide it out from under them.
      touched.current = true;
      const next = clampPage(index, count);
      // A slide, or a cut under Reduce Motion. The user's own swipe is never affected.
      ref.current?.scrollTo({ x: next * width, y: 0, animated: !reduceMotion });
      if (reduceMotion) scrollX.value = next * width;
      setPage(next);
    },
    [count, width, reduceMotion, ref, scrollX],
  );

  const adjust = useCallback((action: string) => goTo(adjustedPage(action, page, count)), [goTo, page, count]);

  const scrollHandler = useAnimatedScrollHandler({
    onScroll: (event) => {
      scrollX.value = event.contentOffset.x;
    },
  });

  const onScrollBeginDrag = useCallback(() => {
    touched.current = true;
  }, []);

  const onMomentumScrollEnd = useCallback(
    (event: NativeSyntheticEvent<NativeScrollEvent>) => setPage(pageAtOffset(event.nativeEvent.contentOffset.x, width, count)),
    [width, count],
  );

  const { peekReady } = options;
  useEffect(() => {
    if (
      !shouldPeek({
        count,
        ready: peekReady,
        peeked: peeked.current,
        touched: touched.current,
        reduceMotion,
        screenReader,
      })
    ) {
      return;
    }
    let out = false;
    let back: ReturnType<typeof setTimeout> | undefined;
    const start = setTimeout(() => {
      if (touched.current) return;
      // Marked when it actually runs, so an effect torn down before its timer (Strict Mode's
      // double run) does not use up the one peek.
      peeked.current = true;
      out = true;
      ref.current?.scrollTo({ x: PEEK_OFFSET, y: 0, animated: true });
      back = setTimeout(() => {
        out = false;
        if (!touched.current) ref.current?.scrollTo({ x: 0, y: 0, animated: true });
      }, PEEK_HOLD_MS);
    }, PEEK_DELAY_MS);
    return () => {
      clearTimeout(start);
      if (back) clearTimeout(back);
      // Torn down mid-peek (Reduce Motion switched on, the screen closing): never leave the
      // pager parked 32pt off a page.
      if (out && !touched.current) ref.current?.scrollTo({ x: 0, y: 0, animated: false });
    };
  }, [count, peekReady, reduceMotion, screenReader, ref]);

  return {
    count,
    width,
    page: clampPage(page, count),
    scrollX,
    ref,
    goTo,
    adjust,
    scrollHandler,
    onScrollBeginDrag,
    onMomentumScrollEnd,
  };
}

/**
 * DuoQueue+ plan maths and wording for the paywall's plans page. Pure (no React Native,
 * relative imports only), so vitest checks every number and sentence the purchase page shows.
 * The shapes below are the parts of react-native-purchases' PurchasesPackage this file reads; a
 * real package satisfies them structurally.
 */

export interface PlanIntroPrice {
  price: number;
  periodUnit: string;
  periodNumberOfUnits: number;
}

export interface PlanProduct {
  identifier: string;
  price: number;
  priceString: string;
  currencyCode: string;
  introPrice: PlanIntroPrice | null;
}

export interface PlanPackage {
  identifier: string;
  packageType: string;
  product: PlanProduct;
}

export type PlanPeriod = "week" | "month" | "threeMonths" | "sixMonths";

/** Shortest first: the order the rows are drawn in (all four stay reachable, Guideline 2.1(b)). */
export const PLAN_ORDER: readonly PlanPeriod[] = ["week", "month", "threeMonths", "sixMonths"];

/** Weeks in each billing period. A month is 52/12 weeks, never 4, so a plan's per-week price
 * and its saving against Weekly come from the same arithmetic. */
export const WEEKS_PER_PERIOD: Record<PlanPeriod, number> = { week: 1, month: 52 / 12, threeMonths: 13, sixMonths: 26 };

/** RevenueCat's INTRO_ELIGIBILITY_STATUS_ELIGIBLE. Unknown (0), ineligible (1) and "no intro
 * offer" (3) all mean no trial is shown. */
export const INTRO_ELIGIBLE = 2;

const PACKAGE_TYPE_PERIOD: Record<string, PlanPeriod> = {
  WEEKLY: "week",
  MONTHLY: "month",
  THREE_MONTH: "threeMonths",
  SIX_MONTH: "sixMonths",
};

/** The product ids in App Store Connect (docs/APP-STORE-SUBMISSION.md Part 2.5). */
const PRODUCT_ID_PERIOD: Record<string, PlanPeriod> = {
  duoqueue_plus_weekly: "week",
  duoqueue_plus_monthly: "month",
  duoqueue_plus_3mo: "threeMonths",
  duoqueue_plus_6mo: "sixMonths",
};

/**
 * RevenueCat's package type first, then our own product id, then null. Never a guess: a wrong
 * period on a row ("Monthly · $29.99") is a refund, a review and a 3.1.2 rejection.
 */
export function planPeriodOf(pkg: Pick<PlanPackage, "packageType" | "product">): PlanPeriod | null {
  return PACKAGE_TYPE_PERIOD[pkg.packageType] ?? PRODUCT_ID_PERIOD[pkg.product.identifier] ?? null;
}

export interface Plan<P extends PlanPackage = PlanPackage> {
  pkg: P;
  period: PlanPeriod;
}

/** The store's packages as plans, shortest first; empty slots, unknown products and a second
 * package for the same period are dropped. */
export function orderPlans<P extends PlanPackage>(pkgs: readonly (P | null | undefined)[]): Plan<P>[] {
  const byPeriod = new Map<PlanPeriod, Plan<P>>();
  for (const pkg of pkgs) {
    if (!pkg) continue;
    const period = planPeriodOf(pkg);
    if (period && !byPeriod.has(period)) byPeriod.set(period, { pkg, period });
  }
  return PLAN_ORDER.flatMap((period) => {
    const plan = byPeriod.get(period);
    return plan ? [plan] : [];
  });
}

/** The picked plan, else Monthly (the default, with no badge), else the first. */
export function selectedPlan<P extends PlanPackage>(plans: readonly Plan<P>[], selectedId: string | null): Plan<P> | null {
  return (
    plans.find((p) => p.pkg.identifier === selectedId) ?? plans.find((p) => p.period === "month") ?? plans[0] ?? null
  );
}

export function pricePerWeek(price: number, period: PlanPeriod): number {
  return price / WEEKS_PER_PERIOD[period];
}

/** A price in the store's currency; the code and two decimals if the platform cannot format it. */
export function formatMoney(amount: number, currencyCode: string, locale?: string): string {
  try {
    return new Intl.NumberFormat(locale, { style: "currency", currency: currencyCode }).format(amount);
  } catch {
    return `${currencyCode} ${amount.toFixed(2)}`;
  }
}

/**
 * Whole percent saved per week against the Weekly plan, rounded DOWN so it never claims more
 * than the store's prices give (76.9% reads "Save 76%"). Null for Weekly itself, without a
 * Weekly plan, or when there is no saving.
 */
export function savingsPercent(plan: Plan, weekly: Plan | null): number | null {
  if (plan.period === "week" || !weekly || weekly.pkg.product.price <= 0) return null;
  const pct = Math.floor((1 - pricePerWeek(plan.pkg.product.price, plan.period) / weekly.pkg.product.price) * 100);
  return pct > 0 ? pct : null;
}

const TRIAL_UNITS: Record<string, readonly [string, string]> = {
  DAY: ["day", "days"],
  WEEK: ["week", "weeks"],
  MONTH: ["month", "months"],
  YEAR: ["year", "years"],
};

/**
 * "7 days", "1 week": a trial to advertise, only for a FREE intro offer (price 0) that RevenueCat
 * says this Apple Account is still eligible for. Anything else is null, so a paid intro price or
 * an unknown eligibility never reads as free (build 17 fix: "Free trial included" showed for any
 * intro price).
 */
export function trialLength(intro: PlanIntroPrice | null, eligibility: number | undefined): string | null {
  if (!intro || intro.price !== 0 || eligibility !== INTRO_ELIGIBLE) return null;
  const unit = TRIAL_UNITS[intro.periodUnit];
  const n = intro.periodNumberOfUnits;
  if (!unit || n <= 0) return null;
  return `${n} ${n === 1 ? unit[0] : unit[1]}`;
}

 * RevenueCat's intro-offer eligibility per product id (its INTRO_ELIGIBILITY_STATUS, as a
 * number). Any failure is an empty map: unknown counts as not eligible, so a failed check shows
 * the plain price rather than a trial the store then refuses (pricing.ts trialLength).

export function useIntroEligibility(productIds: readonly string[]) {
  const key = [...productIds].sort().join(",");
  return useQuery({
    queryKey: ["revenuecat-intro-eligibility", key],
    queryFn: async (): Promise<Record<string, number>> => {
      try {
        const result = await loadPurchases().checkTrialOrIntroductoryPriceEligibility([...productIds]);

interface PurchasesCancelledError {
  userCancelled?: boolean;
}

function wasCancelledByUser(err: unknown): boolean {
  return typeof err === "object" && err !== null && (err as PurchasesCancelledError).userCancelled === true;
}

/**
 * What buying did. Cancelling Apple's sheet is not an error and not a reason to leave: the
 * paywall stays open for it (build 17 fix: it used to close). `active` says whether DuoQueue+ is
 * already on for this account.
 */
export type PurchaseOutcome = { kind: "cancelled" } | { kind: "purchased"; active: boolean };

export function usePurchasePackage() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async (pkg: PurchasesPackage): Promise<PurchaseOutcome> => {
      try {
        const result = await loadPurchases().purchasePackage(pkg);
        return { kind: "purchased", active: hasPremium(result.customerInfo) };
      } catch (err) {
        if (wasCancelledByUser(err)) return { kind: "cancelled" };
        throw err;
      }
    },
    onSuccess: (outcome) => {
      if (outcome.kind === "cancelled") return;
      void queryClient.invalidateQueries({ queryKey: ["subscription"] });
      void queryClient.invalidateQueries({ queryKey: CONSUMABLE_CREDITS_QUERY_KEY });
      // The deck reads its "N swipes left today" label and its is_premium flag from this cache.
      // Without this, somebody who upgrades BECAUSE they ran out lands straight back on a screen
      // still telling them they have none left, which is the worst moment in the app to be wrong.
      void queryClient.invalidateQueries({ queryKey: SWIPE_QUOTA_QUERY_KEY });
    },
  });
}

 * "Restored" and closed). */

      void queryClient.invalidateQueries({ queryKey: ["subscription"] });
      // Restoring on a new device grants the same entitlement a purchase does, so it has to
      // refresh the same caches.

  });

export interface PowerUpState {
  left: number;
  price: string | null;
  busy: boolean;
  buy: () => void;
}

/** The Power-Up purchase, with its own mutation so buying one never shows the subscription
 * button as busy, or the other way round (build 17 fix). A cancelled sheet does nothing. */
export function usePowerUp(): PowerUpState {
  const { data: offering } = useOfferings();
  const { data: credits } = useConsumableCredits();
  const purchase = usePurchasePackage();
  const pkg = offering?.availablePackages.find((p) => p.product.identifier === POWER_UP_PRODUCT_ID) ?? null;

  async function buy() {
    if (!pkg) return;
    try {
      const outcome = await purchase.mutateAsync(pkg);
      if (outcome.kind === "cancelled") return;
      Alert.alert(POWER_UP_BOUGHT.title, POWER_UP_BOUGHT.message);
    } catch (err) {
      Alert.alert("Purchase failed", err instanceof Error ? err.message : "Please try again.");
    }
  }

  return {
    left: credits?.boosts ?? 0,
    price: pkg?.product.priceString ?? null,
    busy: purchase.isPending,
    buy: () => void buy(),
  };
}

import { useIntroEligibility, useOfferings, usePurchasePackage, useRestorePurchases } from "@/features/premium/useOfferings";

/** Page 3's data and purchases. PlansView draws it. */
export function PlansPage({ bottomInset }: { bottomInset: number }) {
  const { data: offering, isLoading, error } = useOfferings();
  const subscribe = usePurchasePackage();
  const restore = useRestorePurchases();
  const powerUp = usePowerUp();
  const [selectedId, setSelectedId] = useState<string | null>(null);

  const plans = useMemo(
    () => orderPlans([offering?.weekly, offering?.monthly, offering?.threeMonth, offering?.sixMonth]),
    [offering],
  );
  const productIds = useMemo(() => plans.map((p) => p.pkg.product.identifier), [plans]);
  const { data: eligibility } = useIntroEligibility(productIds);
  const weekly = plans.find((p) => p.period === "week") ?? null;
  const rows = plans.map((plan) => ({
    id: plan.pkg.identifier,
    copy: planCopy(plan, {
      weekly,
      trial: trialLength(plan.pkg.product.introPrice, eligibility?.[plan.pkg.product.identifier]),
    }),
  }));
  const selected = selectedPlan(plans, selectedId);
  const selectedCopy = rows.find((row) => row.id === selected?.pkg.identifier)?.copy ?? null;

  async function handleSubscribe() {
    if (!selected) return;
    try {
      const outcome = await subscribe.mutateAsync(selected.pkg);
      // Cancelling Apple's sheet is not a failure and not a reason to leave (build 17 fix).
      if (outcome.kind === "cancelled") return;
      const alert = purchaseAlert(outcome.active);
      Alert.alert(alert.title, alert.message, [{ text: "OK", onPress: goBack }]);
    } catch (err) {
      Alert.alert("Purchase failed", err instanceof Error ? err.message : "Please try again.");
    }
  }

  async function handleRestore() {
    try {
      const { active } = await restore.mutateAsync();
      const alert = restoreAlert(active);
      // Only a real restore leaves the paywall; "Nothing to restore" keeps the plans on screen.
      Alert.alert(alert.title, alert.message, active ? [{ text: "OK", onPress: goBack }] : undefined);
    } catch (err) {
      Alert.alert("Restore failed", err instanceof Error ? err.message : "Please try again.");
    }
  }

  return (
    <PlansView
      state={isLoading ? "loading" : error || plans.length === 0 ? "unavailable" : "ready"}
      rows={rows}
      selectedId={selected?.pkg.identifier ?? null}
      selectedCopy={selectedCopy}
      onSelect={setSelectedId}
      subscribing={subscribe.isPending}
      onSubscribe={() => void handleSubscribe()}
      restoring={restore.isPending}
      onRestore={() => void handleRestore()}
      powerUp={powerUp}
      onContinueFree={goBack}
      bottomInset={bottomInset}
    />
  );
}

/**
 * The tour's once-per-account flag. A local flag keyed by the user id, no migration (spec §2).
 * The store is passed in (AsyncStorage in app/welcome.tsx) so this stays pure and tested.
 */
export interface FlagStore {
  getItem(key: string): Promise<string | null>;
  setItem(key: string, value: string): Promise<void>;
}

export function tourSeenKey(userId: string): string {
  return `duoqueue.tour-seen.v1.${userId}`;
}

/** A store that cannot be read counts as seen: skipping the tour is harmless, showing it twice
 * is not. */
export async function hasSeenTour(store: FlagStore, userId: string): Promise<boolean> {
  try {
    return (await store.getItem(tourSeenKey(userId))) === "1";
  } catch {
    return true;
  }
}

export async function markTourSeen(store: FlagStore, userId: string): Promise<void> {
  try {
    await store.setItem(tourSeenKey(userId), "1");
  } catch {
    // Nothing to do: a flag that did not stick can only mean the tour shows once more, and the
    // route is reached only from the end of profile setup anyway.
  }
}

export function tourRank(games: readonly TourGame[]): LolRank | null {
  for (const game of games) {
    if (isLolDivisionalTier(game.lolTier) && game.lolDivision !== null) {
      return { tier: game.lolTier, division: game.lolDivision };
    }
  }
  return null;
}

  const shown = useSharedValue(1);

    if (reduced) return;

    paddingVertical: 5,

      accessibilityElementsHidden
      importantForAccessibility="no-hide-descendants"

export default function WelcomeScreen() {

  const insets = useSafeAreaInsets();
  const { height } = useWindowDimensions();
  const userId = useSessionStore((s) => s.session?.user.id);
  const isActive = useSessionStore((s) => s.profile?.is_active ?? true);
  const refreshProfile = useSessionStore((s) => s.refreshProfile);
  const [seen, setSeen] = useState<boolean | null>(null);

  // The facts that decide the pages, from the reads the Profile tab makes. The wizard's submit
  // already set onboarding_completed on the server, so get_profile_card returns this card.

  const { data: photos } = useOwnProfilePhotos(userId);
  const factsReady = !details.isPending && (!isActive || !card.isPending);

  const games = details.data?.games ?? [];
  const pages = tourPages({ region: card.data?.region ?? null, games });
  const rank = tourRank(games);
  const rankLabel = pages.includes("canDuo") && rank ? formatLolRank(rank) : null;
  const pictureHeight = tourPictureHeight(height);

  const skipTour = !userId || seen === true;
  const ready = !skipTour && seen === false && factsReady;
  const pager = usePager(pages.length, { peekReady: ready });

  useEffect(() => {
    hapticSuccess();

    if (!userId) return;
    let alive = true;
    void hasSeenTour(AsyncStorage, userId).then((value) => {
      if (alive) setSeen(value);
    });
    return () => {
      alive = false;
    };

    if (finishing.current) return;
    finishing.current = true;

    // (onboarding_completed still false in the store) back into onboarding.
    await refreshProfile();
    router.replace("/(tabs)");
  }, [userId, refreshProfile]);

  useEffect(() => {
    if (skipTour) void finish();
