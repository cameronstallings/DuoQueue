## Global Constraints

Every task's requirements include this section.

- **No em dashes** (U+2014) in any user-facing string: screen copy, toasts, alerts, accessibility labels, store copy, review notes, privacy text. Code comments may keep them.
- **The lit rule:** on a card, chartreuse means only "you two fit here". The only chartreuse on a card is the lit Solo/Duo cell (2pt `volt` edge bar and the solid CAN DUO tag) and the existing DUO drag stamp. Page track, IN COMMON, labels and links on the card are never volt.
- **Card-scoped tokens only:** `cardHero`, `cardRecessed`, `cardEdge`, `cardPanel`, `cardMark`, `cardRecessedMuted`, `cardLitFill`, `cardLitInk`. `surface` and `border` do not change app-wide in build 12.
- **Contrast:** card text at least 8.30:1 dark and 6.10:1 Paper, enforced by `pnpm audit:contrast` (Task 1 adds the floor).
- **No migration and no function deploy.** Nothing under `supabase/` changes. Existing gallery rows and gender values stay in the database untouched and unused.
- **Build 11 compatibility rule:** build 11 draws the card only from `header`. It is a manual-release backup and is never released. If it ever is, set `ALSO_WRITE_HEADER = true` in `apps/mobile/src/features/profile/usePhotoUpload.ts` so every picture is also written to `header` until build 11 dies out.
- **One picture everywhere:** every card and pager reads `profileUrl ?? headerUrl` through `cardPicture()`. No reader shows gallery photos.
- **Reduce Motion = fades only:** read `useReducedMotion()` from react-native-reanimated; fades use `.reduceMotion(ReduceMotion.Never)`; no translation, scale or overshoot when it is on.
- **Hands off:** do not edit `site/`, `scripts/build-site.mjs` or `docs/legal/` (another workflow owns them). Privacy 1.4 is written as a draft file outside the repo (Task 12).
- **No dependency changes.** Stay on Expo SDK 54. No `pnpm install`.
- **Pure modules under test** import relatively (never `@/`, which Vite does not resolve). `import type` is fine. `@duoqueue/shared-types` resolves.
- **Line endings:** many files are CRLF in the working tree (`src/theme/tokens.ts`, `app/invited.tsx`, and others). Edit with the Edit tool, which preserves them. Never sed or python regex rewrites. Write new files with the Write tool.
- **Commits:** one commit per task, staging only the files that task touched, by explicit path (quote paths containing `(` `)` `[` `]`). Never `git add -A` or `git add .` (the tree carries unrelated `marketing/`, root scratch files and another workflow's edits). Every message ends with a blank line and `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- **Gates:** run a gate and read its exit code on its own line (`cmd; echo EXIT=$?`). Never `cmd | tail && git commit`.

import type { ModerationStatus } from "@duoqueue/shared-types";

/**
 * What your own card shows in the picture slot, and the tag and line that explain it.
 *
 * Strangers only ever get approved rows (public_profile_media). The owner can read their own
 * rows with moderation_status, so their card can say what strangers are seeing instead. A
 * rejected picture is never drawn, not even for its owner. An approved old cover (`header`,
 * which build 11 and earlier uploaded) stands in for a missing picture, exactly as cardPicture
 * does for strangers.
 *
 * Pure, relative imports only, so it is unit-tested (pictureStatus.test.ts).
 */

export const PICTURE_TAG = {
  add: "ADD A PICTURE",
  review: "PICTURE IN REVIEW",
  rejected: "PICTURE NOT APPROVED",
} as const;

export const PICTURE_NOTE = {
  review: "Your picture is in review. Until it's approved, players see your initial.",
  rejected: "That picture wasn't approved. Pick another one.",
} as const;

export interface OwnPhotoRows {
  profileUrl: string | null;
  profileStatus: ModerationStatus | null;
  headerUrl: string | null;
  headerStatus: ModerationStatus | null;
}

export interface OwnPictureState {
  /** What to draw on your own card; null draws the initial. */
  url: string | null;
  /** The solid tag on your card's picture area, or null for none. */
  tag: string | null;
  /** The line under your card, or null for none. */
  note: string | null;
  /** Counts as "a picture" for completeness: anything but missing or rejected. */
  hasPicture: boolean;
}

export function ownPictureState(rows: OwnPhotoRows): OwnPictureState {
  const legacyCover = rows.headerStatus === "approved" ? rows.headerUrl : null;

  switch (rows.profileStatus) {
    case "approved":
      return { url: rows.profileUrl ?? legacyCover, tag: null, note: null, hasPicture: true };
    case "pending":
      return { url: rows.profileUrl, tag: PICTURE_TAG.review, note: PICTURE_NOTE.review, hasPicture: true };
    case "rejected":
      return { url: legacyCover, tag: PICTURE_TAG.rejected, note: PICTURE_NOTE.rejected, hasPicture: legacyCover !== null };
    case null:
      return legacyCover
        ? { url: legacyCover, tag: null, note: null, hasPicture: true }
        : { url: null, tag: PICTURE_TAG.add, note: null, hasPicture: false };
  }
}

/**
 * What a finished card has, in one place: the Edit screen's meter lists every item, and your
 * card's nudge row names the first missing one. "A picture" is one item (build 12 has no cover
 * and no gallery). Pure, relative imports only (completeness.test.ts).

export type CompletenessRoute = "/edit-details" | "/edit-prompts";

export interface CompletenessItem {
  key: CompletenessKey;
  label: string;
  done: boolean;
  route: CompletenessRoute;
}

export interface CompletenessFacts {
  /** ownPictureState(...).hasPicture: a picture that is approved or in review. */
  hasPicture: boolean;
  games: number;
  shows: number;
  platforms: number;
  playstyles: number;
  prompts: number;
}

export function completenessItems(f: CompletenessFacts): CompletenessItem[] {
  return [
    { key: "picture", label: "a picture", done: f.hasPicture, route: "/edit-details" },
    { key: "game", label: "a game", done: f.games > 0, route: "/edit-details" },
    { key: "show", label: "a show", done: f.shows > 0, route: "/edit-details" },
    { key: "platform", label: "a platform", done: f.platforms > 0, route: "/edit-details" },
    { key: "playstyle", label: "a playstyle", done: f.playstyles > 0, route: "/edit-details" },
    { key: "prompts", label: "your prompts", done: f.prompts >= PROMPT_COUNT, route: "/edit-prompts" },
  ];
}

/** The nudge under your card asks for what most changes the card first. The picture is left
 * out: the card's own ADD A PICTURE tag already asks for it. */
const NUDGE_ORDER: CompletenessKey[] = ["game", "prompts", "show", "platform", "playstyle"];

export function cardNudge(items: CompletenessItem[]): CompletenessItem | null {
  for (const key of NUDGE_ORDER) {
    const item = items.find((i) => i.key === key);
    if (item && !item.done) return item;
  }
  return null;
}

export function nudgeText(item: CompletenessItem): string {
  return `Finish your card: add ${item.label}`;
}

/** moderate-photo runs after the upload returns, so a new picture sits in "pending" for a few
 * seconds (longer when it is held for a person to look at). useUpdatePhoto refetches this query

    .from("profile_media")
    .select("storage_path, photo_role, moderation_status")
    .eq("profile_id", profileId)
    .in("photo_role", ["profile", "header"]);

  const rows = (data ?? []) as ProfileMediaRow[];
  const signedUrls = await signPhotoUrls(rows.map((r) => r.storage_path));

import * as base64js from "base64-js";
import * as FileSystem from "expo-file-system/legacy";
import { useMutation, useQueryClient } from "@tanstack/react-query";

import { supabase } from "@/lib/supabase";
import { useToastStore } from "@/store/toast-store";

/** Matches the profile-photos bucket's file_size_limit (0002_storage.sql) and
 * MAX_IMAGE_BYTES in supabase/functions/moderate-photo/sanitize.ts. Checked client-side
 * before upload so a too-large photo fails with one friendly message here, instead of
 * uploading successfully (Storage's own limit rejects it) or uploading and only failing
 * later inside moderate-photo's re-encode step. */
const MAX_PHOTO_BYTES = 5 * 1024 * 1024;

export class PhotoTooLargeError extends Error {
  constructor() {
    super("That photo is too large. Please choose one under 5MB.");
    this.name = "PhotoTooLargeError";
  }
}

/** Exported so onboarding can check a picture the moment it is picked, on the step where it
 * was picked, rather than only when the last step uploads it. */
export async function assertPhotoWithinSizeLimit(uri: string): Promise<void> {
  const info = await FileSystem.getInfoAsync(uri);
  // `exists` narrows FileInfo so `.size` is present; a picker-returned uri that somehow
  // doesn't exist on disk is a different failure the upload call below will surface on
  // its own — this check only guards the size case.
  if (info.exists && info.size > MAX_PHOTO_BYTES) {
    throw new PhotoTooLargeError();
  }
}

/**
 * Build 11 is a manual-release backup that is never released, and it draws the swipe card
 * only from the `header` row. If build 11 is ever released, set this to true so every
 * picture is also written as `header` until build 11 is gone. That is two uploads and two
 * moderate-photo calls per picture; each row keeps its own storage path because moderation
 * runs per row. See the build 12 spec's compatibility rule.
 */
export const ALSO_WRITE_HEADER = false;

/** Uploads a local image URI to Storage and upserts the profile_media row for the given
 * role, then asks moderate-photo to review it. `onReviewed` runs once moderate-photo has
 * answered (approved, rejected or held for a person), so the owner's review tag can refresh
 * then instead of on the next poll. */
export async function uploadProfilePhoto(
  profileId: string,
  uri: string,
  role: "profile" | "header",
  onReviewed?: () => void,
): Promise<void> {
  await assertPhotoWithinSizeLimit(uri);
  const extension = uri.split(".").pop()?.toLowerCase() ?? "jpg";
  const storagePath = `${profileId}/${role}-${Date.now()}.${extension}`;
  const base64 = await FileSystem.readAsStringAsync(uri, { encoding: "base64" });
  const contentType = extension === "png" ? "image/png" : extension === "webp" ? "image/webp" : "image/jpeg";

  const { error: uploadError } = await supabase.storage
    .from("profile-photos")
    .upload(storagePath, base64js.toByteArray(base64), { contentType, upsert: true });
  if (uploadError) throw uploadError;

  // Update-then-insert rather than upsert: clients hold column-level grants here
  // (insert: profile_id/storage_path/photo_role, update: storage_path only), so an
  // upsert's conflict path would try to write columns it isn't allowed to touch.
  // moderation_status is deliberately never sent — it defaults to 'pending' on insert,
  // and a trigger resets it to 'pending' whenever storage_path changes, so only the
  // moderate-photo function can ever approve a photo.
  const { data: updatedRow, error: updateError } = await supabase
    .from("profile_media")
    .update({ storage_path: storagePath })
    .eq("profile_id", profileId)
    .eq("photo_role", role)
    .select("id")
    .maybeSingle();
  if (updateError) throw updateError;

  let mediaId = updatedRow?.id;

  if (!mediaId) {
    const { data: insertedRow, error: insertError } = await supabase
      .from("profile_media")
      .insert({ profile_id: profileId, storage_path: storagePath, photo_role: role })
      .select("id")
      .single();
    if (insertError) throw insertError;
    mediaId = insertedRow.id;
  }

  // Best-effort, non-blocking: a failed moderation check just leaves the photo
  // "pending" (invisible to other users) rather than failing the upload. moderate-photo
  // writes its verdict before it responds, so onReviewed sees the new status.
  supabase.functions
    .invoke("moderate-photo", { body: { mediaId } })
    .then(() => onReviewed?.())
    .catch((err: unknown) => {
      console.warn("moderate-photo invocation failed:", err);
    });
}

/** The one picture a player has. Onboarding, Edit and the Profile tab all upload through here,
 * so the build 11 rule above is one switch. */
export async function uploadPicture(profileId: string, uri: string, onReviewed?: () => void): Promise<void> {
  await uploadProfilePhoto(profileId, uri, "profile", onReviewed);
  if (ALSO_WRITE_HEADER) await uploadProfilePhoto(profileId, uri, "header");
}

export function useUpdatePhoto(profileId: string | undefined) {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({ uri }: { uri: string }) => {
      if (!profileId) throw new Error("No profile loaded.");
      await uploadPicture(profileId, uri, () => {
        // moderate-photo has answered: clear or change the review tag now.
        void queryClient.invalidateQueries({ queryKey: ["own-profile-photos", profileId] });
      });
    },
    onSuccess: () => {
      void queryClient.invalidateQueries({ queryKey: ["own-profile-photos", profileId] });
      useToastStore.getState().showToast("Picture updated");
    },
    onError: (err) => {
      useToastStore.getState().showToast(err instanceof Error ? err.message : "Couldn't update that picture", "error");
    },
  });
}

// Build 12: the card's three fact cells, its pages, and its details line.
// ---------------------------------------------------------------------------------------------

export interface CellLabels {
  skill: Record<SkillLevel, string>;
  platform: Record<Platform, string>;
  playstyle: Record<PlaystyleTag, string>;
}

/**
 * Where the card is shown decides what the pair cells may say:
 *   - deck: get_deck's rows. Can light, and shared_games_count is relative to you.
 *   - party: get_party_deck counts shared games against every party member, so it is not
 *     "games in common with you". No light, no IN COMMON.
 *   - own: your card on the Profile tab. Never lights ("Can duo" is a fact about a pair); the
 *     third cell is your game total.
 */
export type CardCellsMode = { kind: "deck"; sharedGames: number } | { kind: "party" } | { kind: "own"; totalGames: number };

export interface CardCell {
  key: "game" | "duo" | "platforms" | "style" | "common" | "games";
  /** Rendered uppercase (type.tick). */
  label: string;
  value: string;
  /** The Solo/Duo cell when the server said this pair can duo. The only chartreuse on a card. */
  lit: boolean;
  accessibilityLabel: string;
}

export const CELL_COPY = {
  leagueRank: "League rank",
  soloDuo: "Solo/Duo",
  canDuo: "Can duo",
  playsOn: "Plays on",
  style: "Style",
  inCommon: "In common",
  noneYet: "None yet",
  games: "Games",
} as const;

type CellFacts = Pick<DeckCard, "topGames" | "lolRank" | "lol_duo_eligible" | "platforms" | "playstyles">;

function gameCell(card: CellFacts, labels: CellLabels): CardCell | null {
  if (card.lolRank) {
    return {
      key: "game",
      label: CELL_COPY.leagueRank,
      value: card.lolRank,
      lit: false,
      accessibilityLabel: `League rank: ${card.lolRank}`,
    };
  }
  const top = card.topGames[0];
  if (!top) return null;
  const value = labels.skill[top.skillLevel];
  return { key: "game", label: top.name, value, lit: false, accessibilityLabel: `${top.name}: ${value}` };
}

/** Known playstyle tags in catalog order; unknown tags are dropped. */
function orderedPlaystyles(tags: string[]): PlaystyleTag[] {
  return PLAYSTYLE_TAGS.filter((tag) => tags.includes(tag));
}

function pairOrSetupCell(card: CellFacts, mode: CardCellsMode, labels: CellLabels): CardCell | null {
  if (mode.kind === "deck" && showsDuoLabel(card)) {
    return {
      key: "duo",
      label: CELL_COPY.soloDuo,
      value: CELL_COPY.canDuo,
      lit: true,
      accessibilityLabel: DUO_ACCESSIBILITY_LABEL,
    };
  }
  if (card.platforms.length > 0) {
    const names = card.platforms.map((p) => labels.platform[p]);
    return {
      key: "platforms",
      label: CELL_COPY.playsOn,
      value: names.slice(0, 2).join(" · "),
      lit: false,
      accessibilityLabel: `Plays on: ${names.join(", ")}`,
    };
  }
  const style = orderedPlaystyles(card.playstyles)[0];
  if (style) {
    const value = labels.playstyle[style];
    return { key: "style", label: CELL_COPY.style, value, lit: false, accessibilityLabel: `Style: ${value}` };
  }
  // Never a negative label: with nothing to say, the cell is left out.
  return null;
}

function thirdCell(mode: CardCellsMode): CardCell | null {
  if (mode.kind === "deck") {
    const n = Math.max(0, Math.trunc(mode.sharedGames));
    const value = n === 0 ? CELL_COPY.noneYet : n === 1 ? "1 game" : `${n} games`;
    return {
      key: "common",
      label: CELL_COPY.inCommon,
      value,
      lit: false,
      accessibilityLabel: n === 0 ? "No games in common yet" : `${value} in common`,
    };
  }
  if (mode.kind === "own") {
    return {
      key: "games",
      label: CELL_COPY.games,
      value: String(mode.totalGames),
      lit: false,
      accessibilityLabel: `${mode.totalGames} ${mode.totalGames === 1 ? "game" : "games"}`,
    };
  }
  return null;
}

/**
 * The card's fact cells, left to right:
 *   1. League rank when it can be shown ("League rank / Gold II"), else the top game and its
 *      skill ("Rocket League / Ranked Grinder").
 *   2. The lit Solo/Duo cell when showsDuoLabel() and the mode allows it; else "Plays on"; else
 *      "Style". Never a negative.
 *   3. "In common" on the deck; "Games / n" on your own card; nothing on the party deck.
 */
export function cardCells(card: CellFacts, mode: CardCellsMode, labels: CellLabels): CardCell[] {
  return [gameCell(card, labels), pairOrSetupCell(card, mode, labels), thirdCell(mode)].filter(
    (cell): cell is CardCell => cell !== null,
  );
}

export type CardPage =
  | { kind: "picture"; url: string }
  | { kind: "prompt"; question: string; answer: string }
  | { kind: "blank" };

/**
 * The card's pages: [picture?, prompt 1, prompt 2, prompt 3]. With no picture the card opens on
 * the first prompt; with neither, one blank page that carries the player's initial. Never empty.
 */
export function cardPages(picture: string | null, prompts: DeckCardPrompt[]): CardPage[] {
  const pages: CardPage[] = [];
  if (picture) pages.push({ kind: "picture", url: picture });
  for (const prompt of prompts.slice(0, PROMPT_COUNT)) {
    if (prompt.answer.trim()) pages.push({ kind: "prompt", question: prompt.question, answer: prompt.answer });
  }
  return pages.length > 0 ? pages : [{ kind: "blank" }];
}

/** "27 · NA West · en fr". The card renders it uppercase (type.cardMeta). */
export function cardDetailsLine(
  card: Pick<DeckCard, "age" | "region" | "languages">,
  regionLabels: Record<Region, string>,
): string {
  const parts = [String(card.age), regionLabels[card.region] ?? card.region];
  if (card.languages.length > 0) parts.push(card.languages.join(" "));
  return parts.join(" · ");
}

/**

 * pair into a broken glyph; Array.from walks code points instead. Display names are 2 to 30
 * characters after trim (shared-types displayNameSchema), so "?" only covers bad data.

export function initialOf(name: string | null | undefined): string {
  const first = Array.from((name ?? "").trim())[0];
  return first ? first.toUpperCase() : "?";
}

  FadeIn,
  Keyframe,
  ReduceMotion,
  useAnimatedStyle,
  useReducedMotion,
  useSharedValue,
  withSpring,
  withTiming,
} from "react-native-reanimated";

import { useTheme } from "@/theme/useTheme";

/** Cell widths from the design (124 / 98 / 85 at 345pt), as flex ratios so an SE still fits. */
const CELL_FLEX = [124, 98, 85] as const;
const CELL_STAGGER_MS = 40;
const CELLS_HEIGHT = 40;
const PLATE_INSET = 14;
const TRACK_GAP = 4;
/** Full spring config on purpose: Reanimated 4 merges a partial config over mass 4 /
 * stiffness 900, which wobbles for seconds. Damping ratio ~0.67. */
const PRESS_SPRING = { mass: 1, stiffness: 380, damping: 26 } as const;

/** Each cell prints in: fade plus a 4pt lift, 140ms, 40ms apart. Reduce Motion: a plain fade. */
function cellEntering(index: number, reduced: boolean, duration: number) {
  const delay = index * CELL_STAGGER_MS;
  if (reduced) return FadeIn.duration(duration).delay(delay).reduceMotion(ReduceMotion.Never);
  return new Keyframe({
    0: { opacity: 0, transform: [{ translateY: 4 }] },
    100: { opacity: 1, transform: [{ translateY: 0 }] },
  })
    .duration(duration)
    .delay(delay);
}

/** 2pt segments under the picture; the current one in text ink (not volt) slides over 200ms. */
function PageTrack({ count, index }: { count: number; index: number }) {
  const { colors, motion } = useTheme();
  const [width, setWidth] = useState(0);
  const position = useSharedValue(index);

  useEffect(() => {
    position.value = withTiming(index, { duration: motion.base });
  }, [index, position, motion.base]);

  const segment = count > 0 ? (width - TRACK_GAP * (count - 1)) / count : 0;
  const indicatorStyle = useAnimatedStyle(() => ({
    transform: [{ translateX: position.value * (segment + TRACK_GAP) }],
  }));

  if (count <= 1) return <View style={{ height: 2 }} />;

  return (
    <View
      onLayout={(e) => setWidth(e.nativeEvent.layout.width)}
      style={{ height: 2, flexDirection: "row", gap: TRACK_GAP }}
      accessibilityElementsHidden
      importantForAccessibility="no-hide-descendants"
    >
      {Array.from({ length: count }, (_, i) => (
        <View key={i} style={{ flex: 1, backgroundColor: colors.cardEdge }} />
      ))}
      {width > 0 && (
        <Animated.View
          style={[{ position: "absolute", left: 0, top: 0, height: 2, width: segment, backgroundColor: colors.text }, indicatorStyle]}
        />
      )}
    </View>
  );
}

/** Space a prompt page keeps around its text: 20pt padding each side, a 14pt label, an 8pt gap. */

const PROMPT_LINE = 30;

function CardPageView({ page, name }: { page: CardPage; name: string }) {
  const { colors, spacing, type } = useTheme();
  const [size, setSize] = useState({ width: 0, height: 0 });

  function handleLayout(e: LayoutChangeEvent) {
    const { width, height } = e.nativeEvent.layout;
    setSize({ width, height });
  }

  if (page.kind === "picture") {
    return (
      <Image
        source={{ uri: page.url }}
        style={StyleSheet.absoluteFill}
        contentFit="cover"
        cachePolicy="memory-disk"
        transition={150}
        accessible
        accessibilityRole="image"
        accessibilityLabel={`${name}'s picture`}
      />
    );
  }

  if (page.kind === "blank") {
    const markSize = Math.min(size.width, size.height);
    return (
      <View style={[StyleSheet.absoluteFill, { alignItems: "center", justifyContent: "center" }]} onLayout={handleLayout}>
        {markSize > 0 && <PlayerMark name={name} size={markSize} shape="none" color={colors.cardMark} glyphScale={0.8} />}
      </View>
    );
  }

  const markSize = size.height;
  const answerLines = size.height > 0 ? Math.max(2, Math.min(6, Math.floor((size.height - PROMPT_CHROME) / PROMPT_LINE))) : 6;
  return (
    <View style={StyleSheet.absoluteFill} onLayout={handleLayout}>
      {markSize > 0 && (
        <View pointerEvents="none" style={{ position: "absolute", right: -markSize * 0.15, bottom: -markSize * 0.3 }}>
          <PlayerMark name={name} size={markSize} shape="none" color={colors.cardMark} glyphScale={0.9} />
        </View>
      )}
      <View style={{ padding: spacing.lg - spacing.xs, gap: spacing.sm }}>
        {page.question ? (
          <Text style={[type.label, { color: colors.cardRecessedMuted }]} numberOfLines={2}>
            {page.question}
          </Text>
        ) : null}
        <Text style={[type.promptHero, { color: colors.text }]} numberOfLines={answerLines}>
          {page.answer}
        </Text>
      </View>
    </View>
  );
}

export function CardFace({

import type { SwipeQuota } from "@duoqueue/shared-types";

export type StripQuota = Pick<SwipeQuota, "is_premium" | "swipes_limit" | "swipes_used">;

/** The free daily limit when get_swipe_quota sends none (the old Deck caption used the same). */
export const FREE_DAILY_SWIPES = 25;

/** The strip's right-hand text, rendered uppercase. `compact` is for phones narrower than 380pt,
 * where "4 PICKED TODAY" and "18 SWIPES LEFT" do not both fit. */
export function stripQuotaLabel(quota: StripQuota | undefined, compact: boolean): string {
  if (!quota) return "";
  if (quota.is_premium) return compact ? "Unlimited" : "Unlimited swipes";
  const left = Math.max((quota.swipes_limit ?? FREE_DAILY_SWIPES) - quota.swipes_used, 0);
  if (compact) return `${left} left`;
  return `${left} ${left === 1 ? "swipe" : "swipes"} left`;
}

export function highlightsCountLabel(count: number): string {
  return `${count} picked today`;
}

export function leagueRuleLine(games: { lolTier: LolTierChoice | null; lolDivision: LolDivision | null }[]): string | null {

  }

import { PLAYSTYLE_TAGS, type PlaystyleTag, type TiltHandling } from "@duoqueue/shared-types";

/**
 * HOW I PLAY on the Profile tab: the vibe sliders and play window as words on four tiles. You
 * still edit them as sliders; here "50/50/50" reads "Balanced / Mic sometimes / Open to tips"
 * instead of looking unanswered. Pure (howIPlay.test.ts).
 */

export type VibeBand = "low" | "mid" | "high";

/** A 0-100 slider as one of three bands. The middle is wide on purpose: every account starts at
 * 50 (profile_vibe's default), and a small nudge from 50 should still read as the neutral word. */
export function vibeBand(value: number): VibeBand {
  if (!Number.isFinite(value)) return "mid";
  const v = Math.min(100, Math.max(0, value));
  if (v <= 33) return "low";
  if (v >= 67) return "high";
  return "mid";
}

/** Ends taken from the Edit sliders' own labels ("Chill / norms & ARAM" to "Sweaty ranked grind"). */
export const INTENSITY_WORDS: Record<VibeBand, string> = { low: "Chill", mid: "Balanced", high: "Ranked grind" };
export const COMMS_WORDS: Record<VibeBand, string> = { low: "Mostly quiet", mid: "Mic sometimes", high: "Mic on" };
export const COACHING_WORDS: Record<VibeBand, string> = { low: "No tips, please", mid: "Open to tips", high: "Wants coaching" };

/** The tilt answers are 24-25 characters, too long for a half tile, so each splits in two. */
export const TILT_WORDS: Record<TiltHandling, { value: string; caption: string | null }> = {
  stays_calm: { value: "Stays calm", caption: null },
  gets_frustrated_sometimes: { value: "Gets frustrated", caption: "Sometimes" },
  tilts_but_recovers_fast: { value: "Recovers fast", caption: "After a quick tilt" },
  needs_space_after_losses: { value: "Needs space", caption: null },
};

export const TILE_LABELS = { style: "Style", when: "When", comms: "Comms", afterLoss: "After a loss" } as const;

/** The play-window presets (profile-labels PLAY_WINDOW_PRESETS) by name. Their labels carry en
 * dashes and a trailing hours range, so the names are restated here. */
const WINDOW_NAMES: { startHour: number; endHour: number; name: string }[] = [
  { startHour: 5, endHour: 9, name: "Early morning" },
  { startHour: 9, endHour: 12, name: "Morning" },
  { startHour: 12, endHour: 17, name: "Afternoon" },
  { startHour: 17, endHour: 21, name: "Evening" },
  { startHour: 21, endHour: 1, name: "Night" },
  { startHour: 1, endHour: 5, name: "Late night" },
];

export function hourLabel(hour: number): string {
  const h = ((hour % 24) + 24) % 24;
  if (h === 0) return "12 AM";
  if (h < 12) return `${h} AM`;
  if (h === 12) return "12 PM";
  return `${h - 12} PM`;
}

export function playWindowWords(
  startHour: number | null,
  endHour: number | null,
): { value: string; caption: string | null } | null {
  if (startHour === null || endHour === null) return null;
  const hours = `${hourLabel(startHour)} to ${hourLabel(endHour)}`;
  const preset = WINDOW_NAMES.find((w) => w.startHour === startHour && w.endHour === endHour);
  return preset ? { value: preset.name, caption: hours } : { value: hours, caption: null };
}

export interface HowIPlayTile {
  key: "style" | "when" | "comms" | "afterLoss";
  label: string;
  value: string;
  caption: string | null;
  accessibilityLabel: string;
}

export interface HowIPlayInput {
  vibe: { intensity: number; commsStyle: number; coachingPref: number; tiltHandling: TiltHandling } | null;
  playstyles: PlaystyleTag[];
  playWindow: { startHour: number | null; endHour: number | null };
  playstyleLabels: Record<PlaystyleTag, string>;
  tiltLabels: Record<TiltHandling, string>;
}

function spoken(label: string, value: string, caption: string | null): string {
  return caption ? `${label}: ${value}. ${caption}` : `${label}: ${value}`;
}

/** STYLE, WHEN, COMMS, AFTER A LOSS; a tile with no data is left out. */
export function howIPlayTiles(input: HowIPlayInput): HowIPlayTile[] {
  const tiles: HowIPlayTile[] = [];
  const { vibe } = input;

  if (vibe) {
    const value = INTENSITY_WORDS[vibeBand(vibe.intensity)];
    const tags = PLAYSTYLE_TAGS.filter((tag) => input.playstyles.includes(tag))
      .map((tag) => input.playstyleLabels[tag])
      .filter((label) => label.toLowerCase() !== value.toLowerCase())
      .slice(0, 2);
    const caption = tags.length > 0 ? tags.join(" · ") : null;
    tiles.push({ key: "style", label: TILE_LABELS.style, value, caption, accessibilityLabel: spoken(TILE_LABELS.style, value, caption) });
  }

  const when = playWindowWords(input.playWindow.startHour, input.playWindow.endHour);
  if (when) {
    tiles.push({
      key: "when",
      label: TILE_LABELS.when,
      value: when.value,
      caption: when.caption,
      accessibilityLabel: spoken(TILE_LABELS.when, when.value, when.caption),
    });
  }

  if (vibe) {
    const comms = COMMS_WORDS[vibeBand(vibe.commsStyle)];
    const coaching = COACHING_WORDS[vibeBand(vibe.coachingPref)];
    tiles.push({
      key: "comms",
      label: TILE_LABELS.comms,
      value: comms,
      caption: coaching,
      accessibilityLabel: spoken(TILE_LABELS.comms, comms, coaching),
    });
    const tilt = TILT_WORDS[vibe.tiltHandling];
    tiles.push({
      key: "afterLoss",
      label: TILE_LABELS.afterLoss,
      value: tilt.value,
      caption: tilt.caption,
      accessibilityLabel: `After a losing streak: ${input.tiltLabels[vibe.tiltHandling]}`,
    });
  }

  return tiles;
}

import { VB_W } from "../../theme/logo-geometry";

/**
 * The match moment's rules. Pure, relative imports only (matchMoment.test.ts).
 *
 * No RPC returns pair facts for a match (lol_solo_duo_eligible is revoked from clients, and
 * build 12 adds no server work), so the screen that made the match passes what it knows as
 * route params: "duo" when get_deck said the pair can duo, "common" when it knows the count.
 */

/** How far each ring starts from its locked place, in the mark's own viewBox units. The
 * duoqueue.io match card slides the same distance, so the site and the app tell one moment. */
export const RING_TRAVEL_UNITS = 190;

export function ringTravel(markWidth: number): number {
  return RING_TRAVEL_UNITS * (markWidth / VB_W);
}

// Type aliases, not interfaces: expo-router's params want index-signature-compatible objects.
export type MatchRouteParams = { matchId: string; name?: string; photo?: string; duo?: string; common?: string };
export type MatchPushParams = { matchId: string; name: string; photo: string; duo?: "1"; common?: string };

export interface MatchFacts {
  matchId: string;
  name: string;
  /** Their picture (cardPicture), or null. */
  photo: string | null;
  /** showsDuoLabel(card). Only get_deck rows can make this true. */
  canDuo: boolean;
  /** Games you both play where the screen knows it; null where it does not. */
  sharedGames: number | null;
}

export function matchParams(f: MatchFacts): MatchPushParams {
  return {
    matchId: f.matchId,
    name: f.name,
    photo: f.photo ?? "",
    ...(f.canDuo ? { duo: "1" as const } : {}),
    ...(f.sharedGames !== null && f.sharedGames > 0 ? { common: String(Math.trunc(f.sharedGames)) } : {}),
  };
}

export const MATCH_REASON_COPY = { duo: "You can queue Ranked Solo/Duo" } as const;

export interface MatchReason {
  kind: "duo" | "common";
  text: string;
}

/** Facts only, never a score: "You can queue Ranked Solo/Duo", else "3 games in common", else nothing. */
export function matchReason(params: Pick<MatchRouteParams, "duo" | "common">): MatchReason | null {
  if (params.duo === "1") return { kind: "duo", text: MATCH_REASON_COPY.duo };
  const raw = params.common ?? "";
  if (!/^\d+$/.test(raw)) return null;
  const n = Number(raw);
  if (n <= 0) return null;
  return { kind: "common", text: n === 1 ? "1 game in common" : `${n} games in common` };
}

import { StyleSheet, View } from "react-native";
import Animated, { type SharedValue, useAnimatedStyle } from "react-native-reanimated";
import Svg, { Circle, Path } from "react-native-svg";

import { CXA, CXB, CY, LOGO_MARK_ASPECT, mix, NUB, R, SEG_PATH, VB_H, VB_W, VB_X, VB_Y, W } from "@/theme/logo-geometry";
import { useTheme } from "@/theme/useTheme";

import { ringTravel } from "./matchMoment";

interface RingsLockProps {
  width: number;
  /** 0 = the rings apart (the O left, the Q right); 1 = locked in place. */
  progress: SharedValue<number>;
  /** 0 = the O lies over the Q at both crossings (the approach); 1 = the true weave. */
  woven: SharedValue<number>;
}

/**
 * The O+Q mark in three stacked layers so each ring slides on the UI thread (plain transforms,
 * no animated SVG props): the Q ring and nub, the O ring, then the Q's lower-crossing arc (the
 * weave), which travels with the Q. At progress 1 and woven 1 it draws exactly what
 * <Logo variant="mark"/> draws, in the same order and colours. The weave is held back until the
 * rings touch: drawn during the slide, it only partly covers the crossing and leaves a hard cut.
 */
export function RingsLock({ width, progress, woven }: RingsLockProps) {
  const { colors } = useTheme();
  const height = width * LOGO_MARK_ASPECT;
  const travel = ringTravel(width);
  const ink = colors.text;
  const duo = mix(colors.text, colors.background, 0.58);
  const viewBox = `${VB_X} ${VB_Y} ${VB_W} ${VB_H}`;

  const qStyle = useAnimatedStyle(() => ({ transform: [{ translateX: (1 - progress.value) * travel }] }));
  const oStyle = useAnimatedStyle(() => ({ transform: [{ translateX: -(1 - progress.value) * travel }] }));
  const weaveStyle = useAnimatedStyle(() => ({
    opacity: woven.value,
    transform: [{ translateX: (1 - progress.value) * travel }],
  }));

  return (
    <View
      pointerEvents="none"
      accessibilityElementsHidden
      importantForAccessibility="no-hide-descendants"
      style={{ width, height }}
    >
      <Animated.View style={[StyleSheet.absoluteFill, qStyle]}>
        <Svg width={width} height={height} viewBox={viewBox}>
          <Circle cx={CXB} cy={CY} r={R} fill="none" stroke={duo} strokeWidth={W} />
          <Circle cx={NUB.cx} cy={NUB.cy} r={NUB.r} fill={duo} />
        </Svg>
      </Animated.View>
      <Animated.View style={[StyleSheet.absoluteFill, oStyle]}>
        <Svg width={width} height={height} viewBox={viewBox}>
          <Circle cx={CXA} cy={CY} r={R} fill="none" stroke={ink} strokeWidth={W} />
        </Svg>
      </Animated.View>
      <Animated.View style={[StyleSheet.absoluteFill, weaveStyle]}>
        <Svg width={width} height={height} viewBox={viewBox}>
          <Path d={SEG_PATH} fill="none" stroke={duo} strokeWidth={W} />
        </Svg>
      </Animated.View>
    </View>
  );
}
